/* AI IMAGE GENERATION                               */
/* ================================================= */

async function generateImage(){

    const prompt =
    document.getElementById(
    "prompt"
    ).value.trim();

    if(!prompt){

        alert(
        "Enter a prompt first."
        );

        return;
    }

    try{

        const response =
        await fetch(

        "https://picsum.photos/500/500"

        );

        const imageURL =
        response.url;

        document.getElementById(
        "outputImage"
        ).src = imageURL;

    }catch(error){

        console.log(error);

        alert(
        "Image generation failed."
        );
    }
}

/* ================================================= */
/* PRODUCT UPLOAD                                    */
/* ================================================= */

async function uploadProduct(){

    const name =
    document.getElementById(
    "productName"
    ).value;

    const file =
    document.getElementById(
    "productImage"
    ).files[0];

    if(!name || !file){

        alert(
        "Add product name and image."
        );

        return;
    }

    /* ============================================= */
    /* FIREBASE STORAGE                              */
    /* ============================================= */

    if(storage){

        try{

            const storageRef =
            storage.ref(
            "products/" + file.name
            );

            await storageRef.put(file);

            const imageURL =
            await storageRef.getDownloadURL();

            if(db){

                await db.collection(
                "products"
                ).add({

                    name,
                    image:imageURL
                });
            }

            alert(
            "Product uploaded."
            );

        }catch(error){

            console.log(error);

            alert(
            "Upload failed."
            );
        }

    }else{

        alert(
        "Firebase not connected."
        );
    }
}

/* ================================================= */
/* DROP TIMER                                        */
/* ================================================= */

let targetDate =
localStorage.getItem(
"dropDate"
);

function updateDrop(){

    const value =
    document.getElementById(
    "dropDate"
    ).value;

    if(!value){

        alert(
        "Choose a date."
        );

        return;
    }

    localStorage.setItem(
    "dropDate",
    value
    );

    targetDate = value;
}

/* ================================================= */
/* LIVE TIMER                                        */
/* ================================================= */

setInterval(()=>{

    if(!targetDate) return;

    const now =
    new Date().getTime();

    const target =
    new Date(targetDate)
    .getTime();

    const distance =
    target - now;

    if(distance <= 0){

        document.getElementById(
        "timer"
        ).innerHTML =

        "DROP LIVE";

        return;
    }

    const days =
    Math.floor(
    distance /
    (1000*60*60*24)
    );

    const hours =
    Math.floor(
    (distance %
    (1000*60*60*24))
    /
    (1000*60*60)
    );

    const minutes =
    Math.floor(
    (distance %
    (1000*60*60))
    /
    (1000*60)
    );

    const seconds =
    Math.floor(
    (distance %
    (1000*60))
    /1000
    );

    document.getElementById(
    "timer"
    ).innerHTML =

    `${days}D :
    ${hours}H :
    ${minutes}M :
    ${seconds}S`;

},1000);
